document.getElementById("regForm").addEventListener("submit", function(event) {     event.preventDefault(); // stop page reload 
 
    let name = document.getElementById("name").value.trim();     let email = document.getElementById("email").value.trim();     let phone = document.getElementById("phone").value.trim();     let course = document.getElementById("course").value; 
 
    let isValid = true; 
 
    // Name validation     if (name === "") { 
        document.getElementById("nameError").innerText = "Name is required";         isValid = false; 
    } else { 
        document.getElementById("nameError").innerText = ""; 
    } 
 
    // Email validation 
    let emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; 
    if (!emailPattern.test(email)) { 
        document.getElementById("emailError").innerText = "Enter a valid email";         isValid = false; 
    } else { 
        document.getElementById("emailError").innerText = ""; 
    } 
 
    // Phone validation (numeric & 10 digits)     if (!/^\d{10}$/.test(phone)) { 
        document.getElementById("phoneError").innerText = "Phone must be 10 digits";         isValid = false; 
    } else { 
        document.getElementById("phoneError").innerText = ""; 
    } 
 
    // Course validation     if (course === "") { 
        document.getElementById("courseError").innerText = "Select a course";         isValid = false; 
    } else { 
        document.getElementById("courseError").innerText = ""; 
    } 
 
    // Success message 
    if (isValid) { 
        document.getElementById("successMsg").innerText = "Registration Successful!"; 
    } 
}); 
 
// Real-time validation 
document.getElementById("email").addEventListener("input", function() {     let email = this.value; 
    let emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; 
 
    document.getElementById("emailError").innerText =         emailPattern.test(email) ? "" : "Invalid email format"; 
});

Script.js 
// PRODUCT DATA let products = [ 
    { name: "Laptop", price: 45000, category: "Electronics", image: "img1.jpg" }, 
    { name: "Headphones", price: 1500, category: "Electronics", image: "img2.jpg" }, 
    { name: "T-Shirt", price: 500, category: "Clothes", image: "img3.jpg" }, 
    { name: "Shoes", price: 1200, category: "Clothes", image: "img4.jpg" } 
]; 
 
let cart = []; 
 
// DISPLAY PRODUCTS function loadProducts(list) { 
    const container = document.getElementById("productList");     container.innerHTML = ""; 
 
    list.forEach((p, index) => { 
        let card = document.createElement("div"); 
        card.className = "product-card"; 
 
        card.innerHTML = ` 
            <img src="${p.image}" alt="${p.name}"> 
            <h3>${p.name}</h3> 
            <p>Price: ₹${p.price}</p> 
            <p>${p.category}</p> 
            <button class="add-btn" onclick="addToCart(${index})">Add to Cart</button> 
            <button class="details-btn" onclick="viewDetails(${index})">View Details</button> 
        `; 
 
        container.appendChild(card); 
    }); 
} 
 
loadProducts(products); 
 
// FILTERING 
function filterCategory(cat) { 
    const filtered = products.filter(p => p.category === cat);     loadProducts(filtered); 
} 
 
function showAll() { 
    loadProducts(products); 
} 
 
// SORTING function sortPriceLow() {    const sorted = [...products].sort((a, b) => a.price - b.price); 
    loadProducts(sorted); 
} 
 
function sortPriceHigh() { 
    const sorted = [...products].sort((a, b) => b.price - a.price);     loadProducts(sorted); 
} 
 
// ADD TO CART function addToCart(index) { 
    cart.push(products[index]); 
 
    let cartList = document.getElementById("cartList");     let li = document.createElement("li"); 
    li.textContent = products[index].name + " - ₹" + products[index].price;     cartList.appendChild(li); 
} 
 
// VIEW DETAILS POPUP function viewDetails(index) { 
    document.getElementById("popupImg").src = products[index].image;     document.getElementById("popupName").innerText = products[index].name;     document.getElementById("popupPrice").innerText = "Price: ₹" + products[index].price;     document.getElementById("popupCategory").innerText = "Category: " + products[index].category; 
 
    document.getElementById("popup").style.display = "block"; 
} 
 
function closePopup() { 
    document.getElementById("popup").style.display = "none"; 
}

async function getWeather() { 
    const city = document.getElementById("cityInput").value.trim();     const resultDiv = document.getElementById("weatherResult");     const errorMsg = document.getElementById("errorMsg"); 
 
    resultDiv.innerHTML = ""; 
    errorMsg.innerText = ""; 
 
    if (city === "") { 
        errorMsg.innerText = "Please enter a city name.";         return; 
    } 
 
    const apiKey = "YOUR_API_KEY_HERE"; //    Add your OpenWeatherMap API key 
 
    const url = 
        `https://api.openweathermap.org/data/2.5/weather?q=${city}&units=metric&appid=${apiKey}`; 
     try { 
        const response = await fetch(url); 
 
        if (!response.ok) { 
            throw new Error("City not found."); 
        } 
 
        const data = await response.json(); 
 
        const icon = data.weather[0].icon; 
        const iconUrl = `https://openweathermap.org/img/wn/${icon}@2x.png`; 
 
        resultDiv.innerHTML = ` 
            <h2>${data.name}, ${data.sys.country}</h2> 
            <img src="${iconUrl}" alt="Weather Icon"> 
            <h3>${data.weather[0].description.toUpperCase()}</h3> 
            <p><strong>Temperature:</strong> ${data.main.temp} °C</p> 
            <p><strong>Humidity:</strong> ${data.main.humidity}%</p> 
            <p><strong>Wind Speed:</strong> ${data.wind.speed} m/s</p> 
        `; 
 
    } catch (error) { 
        errorMsg.innerText = "Error: " + error.message; 
    } 
}

// Add Task function addTask() { 
    const title = document.getElementById("title").value.trim();     const desc = document.getElementById("desc").value.trim(); 
    const type = document.getElementById("type").value; 
 
    if (title === "" || desc === "") { 
        alert("Please enter both title and description.");         return; 
    } 
 
    let task; 
 
    if (type === "work") { 
        task = new WorkTask(title, desc);     } else if (type === "personal") { 
        task = new PersonalTask(title, desc); 
    } else { 
        task = new Task(title, desc); 
    } 
 
    tasks.push(task);     saveData(); 
    renderTasks(); 
 
    document.getElementById("title").value = ""; 
    document.getElementById("desc").value = ""; 
}

            ${task.description}<br>

            <span class="edit-btn" onclick="editTask(${index})">✏</span>             <span class="delete-btn" onclick="deleteTask(${index})">🗑</span> 
        `; 
 
      list.appendChild(li); 
    }); 
} 
 
// Edit Task function editTask(index) { 
    let newTitle = prompt("Edit Title:", tasks[index].title); 
    let newDesc = prompt("Edit Description:", tasks[index].description);

        tasks[index].description = newDesc; 
 
        saveData(); 
        renderTasks(); 
    } 
}
