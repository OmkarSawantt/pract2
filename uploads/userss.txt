package org.me;

import java.io.IOException;
import java.io.PrintWriter;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet(name = "TemperatureConverterServlet", urlPatterns = {"/convert"})
public class NewServlet extends HttpServlet {

    protected void processRequest(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        response.setContentType("text/html;charset=UTF-8");
        String result = "";
        
        String temperatureStr = request.getParameter("temperature");
        String unit = request.getParameter("unit");
        
        if (temperatureStr != null && !temperatureStr.isEmpty()) {
            try {
                double temperature = Double.parseDouble(temperatureStr);
                
                if ("Celsius".equalsIgnoreCase(unit)) {
                    double fahrenheit = (temperature * 9 / 5) + 32;
                    result = temperature + "°C is " + fahrenheit + "°F";
                } else if ("Fahrenheit".equalsIgnoreCase(unit)) {
                    double celsius = (temperature - 32) * 5 / 9;
                    result = temperature + "°F is " + celsius + "°C";
                } else {
                    result = "Invalid unit. Please specify either 'Celsius' or 'Fahrenheit'.";
                }
            } catch (NumberFormatException e) {
                result = "Please enter a valid number for temperature.";
            }
        }

        try (PrintWriter out = response.getWriter()) {
            out.println("<!DOCTYPE html>");
            out.println("<html>");
            out.println("<head>");
            out.println("<title>Temperature Converter</title>");
            out.println("</head>");
            out.println("<body>");
            out.println("<h1>Temperature Converter</h1>");
            out.println("<form method='post' action='convert'>");
            out.println("Temperature: <input type='text' name='temperature' required/> <br/>");
            out.println("Unit: <select name='unit' required>");
            out.println("<option value='Celsius'>Celsius</option>");
            out.println("<option value='Fahrenheit'>Fahrenheit</option>");
            out.println("</select><br/>");
            out.println("<input type='submit' value='Convert'/>");
            out.println("</form>");
            out.println("<h2>Result: " + result + "</h2>");
            out.println("</body>");
            out.println("</html>");
        }
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }

    @Override
    public String getServletInfo() {
        return "Temperature Converter Servlet";
    }
}

//servlet for calculator
package mihir.me;

import java.io.IOException;
import java.io.PrintWriter;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet(name = "calculator", urlPatterns = {"/calculator"})
public class calculator extends HttpServlet {

    protected void processRequest(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        response.setContentType("text/html;charset=UTF-8");
        String result = "";
        
        // Get parameters from the request
        String num1Str = request.getParameter("num1");
        String num2Str = request.getParameter("num2");
        String operation = request.getParameter("operation");
        
        if (num1Str != null && num2Str != null && operation != null) {
            try {
                double num1 = Double.parseDouble(num1Str);
                double num2 = Double.parseDouble(num2Str);
                
                switch (operation) {
                    case "add":
                        result = num1 + num2 + "";
                        break;
                    case "subtract":
                        result = num1 - num2 + "";
                        break;
                    case "multiply":
                        result = num1 * num2 + "";
                        break;
                    case "divide":
                        if (num2 != 0) {
                            result = num1 / num2 + "";
                        } else {
                            result = "Cannot divide by zero.";
                        }
                        break;
                    default:
                        result = "Invalid operation.";
                }
            } catch (NumberFormatException e) {
                result = "Please enter valid numbers.";
            }
        }

        try (PrintWriter out = response.getWriter()) {
            out.println("<!DOCTYPE html>");
            out.println("<html>");
            out.println("<head>");
            out.println("<title>Calculator Servlet</title>");
            out.println("</head>");
            out.println("<body>");
            out.println("<h1>Simple Calculator</h1>");
            out.println("<form method='post' action='calculator'>");
            out.println("Number 1: <input type='text' name='num1' required/><br/>");
            out.println("Number 2: <input type='text' name='num2' required/><br/>");
            out.println("Operation: <select name='operation' required>");
            out.println("<option value='add'>Add</option>");
            out.println("<option value='subtract'>Subtract</option>");
            out.println("<option value='multiply'>Multiply</option>");
            out.println("<option value='divide'>Divide</option>");
            out.println("</select><br/>");
            out.println("<input type='submit' value='Calculate'/>");
            out.println("</form>");
            out.println("<h2>Result: " + result + "</h2>");
            out.println("</body>");
            out.println("</html>");
        }
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }

    @Override
    public String getServletInfo() {
        return "Simple Calculator Servlet";
    }
}
