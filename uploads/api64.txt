            var number1 = parseFloat(document.getElementById("number1").value);
            var number2 = parseFloat(document.getElementById("number2").value);
            var result = "";
            if (number1 > number2) {
                result = "Number 1 is greater than Number 2";

//	Write a program in java to update 10 records in excel sheet using jxl jar. Include the header for each column. Also display the highest, lowest and average score. 

import java.io.File;
import java.io.IOException;
import jxl.*;
import jxl.read.biff.BiffException;
import jxl.write.*;
import jxl.write.Number;
public class excelClass {
	public static void main(String[] args) {
        String excelFilePath = "C:\\Users\\os521\\OneDrive\\Desktop\\sample.xls";
        try {
        	Workbook workbook = Workbook.getWorkbook(new File(excelFilePath));
            WritableWorkbook copy = Workbook.createWorkbook(new File(excelFilePath), workbook);
            WritableSheet sheet = copy.getSheet(0);
            Label header1 = new Label(0, 0, "Name");
            Label header2 = new Label(1, 0, "Score");
            Label header3 = new Label(2, 0, "Pass");
            sheet.addCell(header1);
            sheet.addCell(header2);
            sheet.addCell(header3);
            double highestScore = Double.MIN_VALUE;
            double lowestScore = Double.MAX_VALUE;
            double totalScore = 0;
            for (int rowIndex = 1; rowIndex <= 10; rowIndex++) {
                Label label1 = new Label(0, rowIndex, "Student " + rowIndex);
                sheet.addCell(label1);
                double score = 50 + rowIndex; 
                Number number = new Number(1, rowIndex, score);
                sheet.addCell(number);
                Label label2 = new Label(2, rowIndex, (rowIndex % 2 == 0) ? "TRUE" : "FALSE");
                sheet.addCell(label2);
                if (score > highestScore) highestScore = score;
                if (score < lowestScore) lowestScore = score;
                totalScore += score;
            }
            double averageScore = totalScore / 10;
            int statsStartRow = 12; 
            Label highestLabel = new Label(0, statsStartRow, "Highest Score:");
            sheet.addCell(highestLabel);
            Number highestValue = new Number(1, statsStartRow, highestScore);
            sheet.addCell(highestValue);
            Label lowestLabel = new Label(0, statsStartRow + 1, "Lowest Score:");
            sheet.addCell(lowestLabel);
            Number lowestValue = new Number(1, statsStartRow + 1, lowestScore);
            sheet.addCell(lowestValue);
            Label averageLabel = new Label(0, statsStartRow + 2, "Average Score:");
            sheet.addCell(averageLabel);
            Number averageValue = new Number(1, statsStartRow + 2, averageScore);
            sheet.addCell(averageValue);
            copy.write();
            copy.close();
            workbook.close();
            System.out.println("Excel file updated successfully!");
        } catch (IOException | BiffException | WriteException e) {
            e.printStackTrace();
        }
	}
}
